CREATE TABLE "user_merch" (
	"id" varchar(50) PRIMARY KEY,
	"user_id" integer NOT NULL,
	"anime_id" varchar(50),
	"item_name" varchar(255) NOT NULL,
	"manufacturer" varchar(255),
	"merch_type" varchar(50) NOT NULL,
	"cost" double precision NOT NULL,
	"bought_from" varchar(255),
	"status" varchar(50) NOT NULL,
	"image_url" text,
	"created_at" timestamp with time zone NOT NULL
);
CREATE UNIQUE INDEX "user_merch_pkey" ON "user_merch" ("id");
ALTER TABLE "user_merch" ADD CONSTRAINT "user_merch_anime_id_fkey" FOREIGN KEY ("anime_id") REFERENCES "anime"("id");
ALTER TABLE "user_merch" ADD CONSTRAINT "user_merch_user_id_fkey" FOREIGN KEY ("user_id") REFERENCES "users"("id");
