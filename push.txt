// Helper function to calculate user level and progress percentage
function calculateUserLevel(xp) {
  const levels = [
    { level: 1, title: 'Otaku Novice', minXP: 0, nextLevelXP: 500 },
    { level: 2, title: 'Casual Watcher', minXP: 500, nextLevelXP: 2000 },
    { level: 3, title: 'Anime Enthusiast', minXP: 1200, nextLevelXP: 3000 },
    { level: 4, title: 'Seasoned Binger', minXP: 2500, nextLevelXP: 5000 },
    { level: 5, title: 'Master Otaku', minXP: 5000, nextLevelXP: 10000 },
    { level: 6, title: 'Anime Scholar', minXP: 10000, nextLevelXP: 20000 },
    { level: 7, title: 'Supreme Weeb', minXP: 20000, nextLevelXP: 50000 }
  ];

  for (let i = levels.length - 1; i >= 0; i--) {
    if (xp >= levels[i].minXP) {
      const lvl = levels[i];
      const range = lvl.nextLevelXP - lvl.minXP;
      const progress = xp - lvl.minXP;
      const percent = Math.min(100, Math.max(0, Math.floor((progress / range) * 100)));
      return {
        level: lvl.level,
        title: lvl.title,
        currentXP: xp,
        nextLevelXP: lvl.nextLevelXP,
        percent: percent
      };
    }
  }
  return { level: 1, title: 'Otaku Novice', currentXP: xp, nextLevelXP: 500, percent: 0 };
}
